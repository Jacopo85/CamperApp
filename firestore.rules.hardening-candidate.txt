// BOZZA PER REVISIONE - NON PUBBLICATA SU FIREBASE.
// NON SOSTITUIRE firestore.rules IN PRODUZIONE prima di verificare le regole
// realmente attive ed eseguire test in Firebase Emulator / Rules Playground.
// Sono cambiati SOLO i permessi di UPDATE per gruppi e membri.
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function isMember(groupId) {
      return signedIn() &&
        exists(/databases/$(database)/documents/groups/$(groupId)/members/$(request.auth.uid));
    }
    match /users/{userId} {
      allow read: if signedIn() && request.auth.uid == userId;
      allow create, update: if signedIn() && request.auth.uid == userId
        && request.resource.data.keys().hasOnly(['email','groupId','updatedAt']);
      allow delete: if false;
    }
    match /invites/{code} {
      // Il codice è casuale e lungo; la lettura è necessaria per consentire l'ingresso.
      allow get: if signedIn();
      allow list: if false;
      allow create: if signedIn()
        && request.resource.data.createdBy == request.auth.uid
        && request.resource.data.groupId is string
        && exists(/databases/$(database)/documents/groups/$(request.resource.data.groupId))
        && get(/databases/$(database)/documents/groups/$(request.resource.data.groupId)).data.ownerUid == request.auth.uid;
      allow update, delete: if false;
    }
    match /groups/{groupId} {
      allow create: if signedIn() && request.resource.data.ownerUid == request.auth.uid;
      allow read: if isMember(groupId);
      // CANDIDATO NON DISTRIBUITO: solo il proprietario puo cambiare il nome.
      allow update: if isMember(groupId)
        && request.auth.uid == resource.data.ownerUid
        && request.resource.data.ownerUid == resource.data.ownerUid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['name'])
        && request.resource.data.name is string
        && request.resource.data.name.size() > 0
        && request.resource.data.name.size() <= 100;
      allow delete: if false;
      match /members/{userId} {
        allow read: if isMember(groupId);
        allow create: if signedIn() && request.auth.uid == userId
          && (
            (request.resource.data.role == 'owner'
             && get(/databases/$(database)/documents/groups/$(groupId)).data.ownerUid == request.auth.uid)
            ||
            (request.resource.data.role == 'member'
             && request.resource.data.inviteCode is string
             && get(/databases/$(database)/documents/invites/$(request.resource.data.inviteCode)).data.groupId == groupId)
          );
        // CANDIDATO NON DISTRIBUITO: l'app non dispone di modifiche ai ruoli.
        // Impedisce agli account membri di auto-promuoversi a owner.
        allow update: if false;
        allow delete: if signedIn() && request.auth.uid == userId && resource.data.role != 'owner';
      }
      match /{subcollection}/{documentId} {
        allow read, create, update, delete: if isMember(groupId)
          && subcollection in ['shopping','wishlist','settings'];
      }
    }
  }
}
